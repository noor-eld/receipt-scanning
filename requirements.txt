# requirements.txt
# These are all the Python libraries the backend needs.

fastapi          # The web framework — handles incoming requests from the phone
uvicorn          # The server that runs FastAPI
python-multipart # Lets FastAPI receive uploaded files (like receipt images)
requests         # Makes HTTP calls to Google Cloud Vision API
python-dotenv    # Reads the .env file
Pillow           # Image processing library (resizing, converting formats)
scikit-learn       # ML library — TF-IDF, Logistic Regression, cross-validation
pandas             # Data handling — reads the training CSV
rapidfuzz           # Fuzzy string matching for ingredient normalization
